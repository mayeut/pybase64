cmake_minimum_required(VERSION 3.22...4.4)
project(${SKBUILD_PROJECT_NAME} LANGUAGES C CXX)

include(CMakeDependentOption)
include(CheckCXXCompilerFlag)
include(CheckLinkerFlag)

check_cxx_compiler_flag("-fno-exceptions" HAVE_CXX_FLAG_NO_EXCEPTIONS)
check_cxx_compiler_flag("-fno-rtti" HAVE_CXX_FLAG_NO_RTTI)
check_linker_flag(CXX "-nostdlib++" HAVE_CXX_LINKER_FLAG_NO_STDLIBXX)
check_linker_flag(CXX "-static-libgcc" HAVE_CXX_LINKER_FLAG_STATIC_LIBGCC)
check_linker_flag(CXX "-static-libstdc++" HAVE_CXX_LINKER_FLAG_STATIC_STDLIBXX)

cmake_dependent_option(PYBASE64_NO_STDLIBXX "disable libstdc++" OFF "HAVE_CXX_FLAG_NO_EXCEPTIONS AND HAVE_CXX_FLAG_NO_RTTI AND HAVE_CXX_LINKER_FLAG_NO_STDLIBXX" OFF)
cmake_dependent_option(PYBASE64_STATIC_LINK "statically link support libraries" OFF "HAVE_CXX_LINKER_FLAG_STATIC_LIBGCC AND (HAVE_CXX_LINKER_FLAG_STATIC_STDLIBXX OR PYBASE64_NO_STDLIBXX)" OFF)

if(CMAKE_CROSSCOMPILING)
  find_package(Python COMPONENTS Development.Module REQUIRED)
else()
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(BUILD_SHARED_LIBS OFF CACHE INTERNAL "" FORCE)

set(BASE64_BUILD_CLI OFF CACHE INTERNAL "" FORCE)
set(BASE64_BUILD_TESTS OFF CACHE INTERNAL "" FORCE)
set(BASE64_WITH_OpenMP OFF CACHE INTERNAL "" FORCE)
set(BASE64_REGENERATE_TABLES OFF CACHE INTERNAL "" FORCE)
add_subdirectory(vendor/base64 EXCLUDE_FROM_ALL)
target_compile_features(base64 PUBLIC c_std_99)

python_add_library(_pybase64 MODULE src/${SKBUILD_PROJECT_NAME}/_pybase64.cpp src/${SKBUILD_PROJECT_NAME}/_pybase64_get_simd_flags.cpp WITH_SOABI)
if(MSVC)
  target_compile_options(_pybase64 PRIVATE "/EHsc")
  target_compile_features(_pybase64 PUBLIC cxx_std_20)
else()
  target_compile_features(_pybase64 PUBLIC cxx_std_17)
endif()
target_link_libraries(_pybase64 PRIVATE base64)
target_include_directories(_pybase64 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/simdutf" "${CMAKE_CURRENT_BINARY_DIR}/vendor/base64" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/base64/lib")
# disable unused features from simdutf
foreach(FEAATURE IN ITEMS DETECT_ENCODING ASCII LATIN1 UTF8 UTF16 UTF32)
  target_compile_definitions(_pybase64 PRIVATE SIMDUTF_FEATURE_${FEAATURE}=0)
endforeach()
if(PYBASE64_NO_STDLIBXX)
  target_compile_definitions(_pybase64 PRIVATE SIMDUTF_NO_LIBCXX=1)
  target_compile_options(_pybase64 PRIVATE -fno-exceptions -fno-rtti)
  target_link_options(_pybase64 PRIVATE -nostdlib++)
  if(PYBASE64_STATIC_LINK)
    target_link_options(_pybase64 PRIVATE -static-libgcc)
  endif()
elseif(PYBASE64_STATIC_LINK)
  target_link_options(_pybase64 PRIVATE -static-libstdc++ -static-libgcc)
endif()

install(TARGETS _pybase64 DESTINATION ${SKBUILD_PROJECT_NAME})

message(STATUS "PYBASE64_NO_STDLIBXX: ${PYBASE64_NO_STDLIBXX}")
message(STATUS "PYBASE64_STATIC_LINK: ${PYBASE64_STATIC_LINK}")
